Exercise 35 — Ask an integer

    return int(input(question))

# Main program
a = ask_integer("First number: ")
b = ask_integer("Second number: ")
print(a + b)

Exercise 36 — Ask until positive

def ask_positive():
    number = int(input("Give me a positive number: "))
    while number <= 0:
        print("That is not positive. Try again.")
        number = int(input("Give me a positive number: "))
    return number

# Main program
n = ask_positive()
print("Thanks! You gave me", n)

Exercise 37 — Sum until stop

def sum_until_stop():
    total = 0
    x = input("Give me a number or write stop: ")
    while x != "stop":
        total += int(x)
        x = input("Give me a number or write stop: ")
    return total

# Main program
print("Total:", sum_until_stop())

Exercise 38 — Average until stop

def average_until_stop():
    total = 0
    count = 0
    x = input("Give me a number or write stop: ")
    while x != "stop":
        total += int(x)
        count += 1
        x = input("Give me a number or write stop: ")
    if count == 0:
        return 0
    return total / count

# Main program
print("Average:", average_until_stop())

Exercise 39 — Password with attempts

def ask_password(correct):
    attempts = 1
    answer = input("Password: ")
    while answer != correct:
        print("Wrong password")
        attempts += 1
        answer = input("Password: ")
    return attempts

# Main program
n = ask_password("python")
print("Welcome! It took you", n, "attempt(s).")

Exercise 40 — Yes or no

def yes_or_no(question):
    answer = input(question + " (yes/no) ")
    while answer != "yes" and answer != "no":
        print("Please answer yes or no.")
        answer = input(question + " (yes/no) ")
    return answer == "yes"

# Main program
if yes_or_no("Do you like Python?"):
    print("Great!")
else:
    print("Keep trying, it gets better!")
