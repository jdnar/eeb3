Mini-project 1: Guess the number (with functions)

import random

def ask_guess():
    return int(input("Guess a number between 1 and 10: "))

def check_guess(secret, guess):
    if guess < secret:
        return "too low"
    elif guess > secret:
        return "too high"
    else:
        return "correct"

def play():
    secret = random.randint(1, 10)
    attempts = 3
    found = False
    while found == False and attempts > 0:
        result = check_guess(secret, ask_guess())
        if result == "correct":
            print("Congratulations, you found the number!")
            found = True
        else:
            print("Wrong: that is", result)
            attempts -= 1
    if found == False:
        print("You have no attempts left. The number was", secret)

# Main program
play()

Mini-project 2: A calculator

def calculate(a, operator, b):
    if operator == "+":
        return a + b
    elif operator == "-":
        return a - b
    elif operator == "*":
        return a * b
    elif operator == "/":
        if b == 0:
            return None
        return a / b
    else:
        return None

# Main program
again = "yes"
while again == "yes":
    a = float(input("First number: "))
    operator = input("Operator (+ - * /): ")
    b = float(input("Second number: "))
    result = calculate(a, operator, b)
    if result == None:
        print("Invalid operation")
    else:
        print("Result:", result)
    again = input("Another one? (yes/no) ")

Mini-project 3: Times-table quiz

import random

def ask_question():
    a = random.randint(1, 10)
    b = random.randint(1, 10)
    answer = int(input(str(a) + " x " + str(b) + " = "))
    if answer == a * b:
        print("Correct!")
        return True
    else:
        print("Wrong, the answer was", a * b)
        return False

def quiz(n):
    score = 0
    for i in range(n):
        if ask_question():
            score += 1
    return score

# Main program
n = int(input("How many questions? "))
score = quiz(n)
print("Score:", score, "/", n)

Mini-project 4: Rock, Paper, Scissors

import random

def random_move():
    return random.choice(["rock", "paper", "scissors"])

def winner(player, computer):
    if player == computer:
        return "draw"
    elif (player == "rock" and computer == "scissors") or (player == "scissors" and computer == "paper") or (player == "paper" and computer == "rock"):
        return "player"
    else:
        return "computer"

def play_round():
    player = input("Rock, paper or scissors? ")
    computer = random_move()
    print("The computer chose", computer)
    result = winner(player, computer)
    if result == "draw":
        print("It's a draw!")
    elif result == "player":
        print("You win!")
    else:
        print("The computer wins!")
    return result

# Main program
player_score = 0
computer_score = 0
play = "yes"
while play == "yes":
    result = play_round()
    if result == "player":
        player_score += 1
    elif result == "computer":
        computer_score += 1
    print("Score - You:", player_score, "Computer:", computer_score)
    play = input("Play again? (yes/no) ")

Mini-project 5: Temperature table

def celsius_to_fahrenheit(celsius):
    return celsius * 9 / 5 + 32

def print_temperature_table(start, stop, step):
    for celsius in range(start, stop + 1, step):
        print(celsius, "C =", celsius_to_fahrenheit(celsius), "F")

# Main program
print_temperature_table(0, 100, 20)
