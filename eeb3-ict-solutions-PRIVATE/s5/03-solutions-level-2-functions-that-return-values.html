    return a + b

Exercise 8 — Perimeter and area

def rectangle_perimeter(length, width):
    return 2 * length + 2 * width

def rectangle_area(length, width):
    return length * width

# Main program
length = float(input("Give me a length: "))
width = float(input("Give me a width: "))
print("Area:", rectangle_area(length, width), "Perimeter:", rectangle_perimeter(length, width))

Exercise 11 — Area of a circle

def circle_area(radius):
    pi = 3.14159
    return pi * radius ** 2

Exercise 12 — Quotient and remainder

def quotient_and_remainder(a, b):
    return (a // b, a % b)

# Main program
first = int(input("Give me a first number: "))
second = int(input("Give me a second number: "))
result = quotient_and_remainder(first, second)
print(first, ":", second, "= quotient", result[0], "and remainder", result[1])

Exercise 13 — Price with tax

def price_with_tax(price, tax):
    return price + price * tax / 100

Exercise 14 — Hours and minutes

def to_hours_minutes(total_minutes):
    return (total_minutes // 60, total_minutes % 60)

# Main program
total = int(input("How many minutes? "))
result = to_hours_minutes(total)
print(total, "minutes =", result[0], "h", result[1], "min")
