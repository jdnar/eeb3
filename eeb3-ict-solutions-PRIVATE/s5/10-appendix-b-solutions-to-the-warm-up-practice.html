B.1 Functions

# 1.1
def say_hello():
    print("Hello!")

say_hello()
say_hello()
say_hello()

# 1.2
def greet(name):
    print("Hello, " + name + "!")

greet("Ana")
greet("Luis")

# 1.3
def rectangle_area(width, height):
    return width * height

print(rectangle_area(3, 5))

# 1.4
def double(n):
    return n * 2

x = double(4)
print(x + 1)

# 1.5  The variable total only exists inside make_total, so print(total)
#      raises a NameError. Print the returned value instead:
def make_total(a, b):
    total = a + b
    return total

print(make_total(2, 3))

# 1.6
def min_max(a, b):
    if a < b:
        return (a, b)
    return (b, a)

r = min_max(9, 4)
print(r[0], r[1])

B.2 Strings

t = "hello world"
print(t.upper())
print(t.lower())
print(t.capitalize())

# 2.2
name = "Marc"
age = 21
print("Hello " + name + " you are " + str(age))

# 2.3
print(len("Python"))

# 2.4  Predicted output for w = "Python"
# w[0]   -> P
# w[-1]  -> n

# w[2:]  -> thon
# w[1:4] -> yth

B.3 Lists

# 3.1
numbers = []
fruits = ["apple", "banana", "cherry"]
print(numbers)
print(fruits)

# 3.2
numbers = []
numbers.append(10)
numbers.append(20)
numbers.insert(1, 15)
print(numbers)              # [10, 15, 20]
numbers.extend([30, 40])
print(numbers)              # [10, 15, 20, 30, 40]

# 3.3
del numbers[0]
print(numbers)              # [15, 20, 30, 40]
numbers.remove(30)
print(numbers)              # [15, 20, 40]

# 3.4
numbers[0] = 99
print(numbers, numbers[1:3])   # [99, 20, 40] [20, 40]

# 3.5
for n in numbers:
    print(n * 2)
print(20 in numbers)        # True

B.4 Split & Join

# 4.1
words = "I love Python".split(" ")
print(words)
print(len(words))

# 4.2
print("-".join(words))

B.5 Dictionaries

# 5.1
pet = {"name": "Rex", "age": 3}
print(pet["name"])

# 5.2
pet["age"] = 4
pet["color"] = "brown"
del pet["age"]
print(pet)                  # {'name': 'Rex', 'color': 'brown'}

# 5.3
print("color" in pet)       # True

# 5.4
for key in pet:
    print(key, pet[key])

B.6 Files

# 6.1
my_file = open("practice.txt", "w")
my_file.write("line one\n")
my_file.close()

# 6.2  (write mode "w" would have erased "line one")
my_file = open("practice.txt", "a")
my_file.write("line two\n")
my_file.close()

# 6.3
my_file = open("practice.txt", "r")
content = my_file.read()
my_file.close()
print(content)

# 6.4  split gives ['line one', 'line two', ''], so remove 1 for the empty end
lines = content.split("\n")
print(len(lines) - 1)
