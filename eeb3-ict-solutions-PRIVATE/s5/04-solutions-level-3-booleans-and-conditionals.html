Exercise 17 — The larger number

def maximum(a, b):
    if a > b:
        return a
    else:
        return b

Exercise 18 — Positive, negative or zero

def sign(n):
    if n > 0:
        return "positive"
    elif n < 0:
        return "negative"
    else:
        return "zero"

Exercise 19 — School grade

def grade(score):
    if score >= 16:
        return "Excellent"
    elif score >= 10:
        return "Pass"
    else:
        return "Fail"

Exercise 20 — Is it in the range?

def in_range(n, low, high):
    return n >= low and n <= high

Exercise 21 — Leap year

def is_leap_year(year):
    return (year % 4 == 0 and year % 100 != 0) or year % 400 == 0

Exercise 22 — Can I buy it?

def can_buy(money, price):
    return money >= price

# Main program
money = float(input("How much money do you have? "))
price = float(input("How much does it cost? "))
if can_buy(money, price):
    print("You can buy it!")
else:
    print("You need", price - money, "more.")

Exercise 23 — The thermostat

def thermostat(current, ideal):
    if current > ideal:
        return "Decrease the temperature."
    elif current < ideal:
        return "Increase the temperature."
    else:
        return "Perfect!"

Exercise 24 — Ticket price

def ticket_price(age):
    if age < 12:
        return 5
    elif age < 18:
        return 8
    elif age < 65:
        return 10
    else:
        return 6
