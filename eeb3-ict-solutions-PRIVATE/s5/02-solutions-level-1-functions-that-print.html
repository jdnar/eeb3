Exercise 2 — A separator line

def separator():
    print("=" * 30)

# Main program
separator()
print("Python Practice")
separator()

Exercise 3 — Greet the user

def greet(name):
    print("Hello, " + name + "!")
    print("Nice to meet you.")

# Main program
name = input("What is your name? ")
greet(name)

Exercise 4 — Introduce a person

def introduce(name, age):
    print(name, "is", age, "years old and will be", age + 1, "next year.")

# Main program
person = input("Name: ")
years = int(input("Age: "))
introduce(person, years)

Exercise 5 — Repeat a text

def repeat(text, times):
    print(text * times)

# Main program
repeat("ha", 3)
repeat("*", 10)
