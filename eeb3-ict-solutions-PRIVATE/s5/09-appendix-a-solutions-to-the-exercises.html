# exercise6
def ingly(word):
    if len(word) > 3:
        if word[-3:] == "ing":
            return word + "ly"
        else:
            return word + "ing"
    return word

word = input("Give me a word: ")
print(ingly(word))
Note: The exercise text says "more than 3 letters" for the first two rules and "less than three letters" for the last one, which leaves words of exactly 3 letters undefined. The solution returns them unchanged.

def largestNumber(list_number):
    largest = list_number[0]
    for number in list_number:
        if number > largest:
            largest = number
    return largest

# exercise12
def occurrences(list_letters, letter):
    count = 0
    indexes = []
    for i in range(len(list_letters)):
        if list_letters[i] == letter:
            count += 1
            indexes.append(i)
    return (count, indexes)

letters = []
askAgain = True
while askAgain == True:
    answer = input("Give me a letter (or write stop if you are done): ")
    if answer == "stop":
        askAgain = False
    else:
        letters.append(answer)

chosen = input("Which letter do you want to count? ")
result = occurrences(letters, chosen)
print("The letter appears", result[0], "times.")
print("It appears at the following index:", result[1])
Note: In Exercise 9 the hint suggests starting the "largest" variable at 0, which fails for lists with only negative numbers. Starting from the first element, as above, works for any non-empty list. Exercises 9, 10 and 11 read their lists with the same while loop as Exercise 8 and then print the value returned by their function.

import random

    my_file = open(file, "r")
    content = my_file.read()
    my_file.close()
    words = content.split("\n")
    return words[random.randint(0, len(words) - 1)]

file = "words.txt"
print(getWord(file))
Note: If your words.txt ends with an empty last line, split("\n") produces an empty string at the end and the program could occasionally pick it. The Hangman solution below removes empty lines to avoid this.

A.7 Project: The Hangman

One possible solution following the four functions of the project:

import random

def getWord(file):
    my_file = open(file, "r")
    content = my_file.read()
    my_file.close()
    words = content.split("\n")
    words = [w for w in words if w != ""]      # ignore empty lines
    return words[random.randint(0, len(words) - 1)]

def addLetter(letter, userWord, word):
    for i in range(len(word)):
        if word[i] == letter:
            userWord[i] = letter
    return userWord

def displayWord(userWord):
    return " ".join(userWord)

def mainGame():
    name = input("What is your name? ")
    word = getWord("words.txt")
    userWord = ["_"] * len(word)
    chances = 8
    alreadyTried = []

    while chances > 0 and "_" in userWord:
        print()
        print(displayWord(userWord))
        letter = input("Choose a letter: ").lower()

        if len(letter) != 1:
            print("Please type exactly one letter.")
        elif letter in alreadyTried:
            print("You already tried this letter.")
        else:
            alreadyTried.append(letter)
            if letter in word:
                userWord = addLetter(letter, userWord, word)
                print("Good one!")
            else:
                chances -= 1
                print("Not in the word.", chances, "chance(s) left.")

    print()
    if "_" not in userWord:
        print("Congratulations", name + "! The word was", word + ".")
        print("Your score:", chances)
    else:
        print("Sorry", name + ", you lost. The word was", word + ".")

mainGame()

Ideas to go further: show the letters already tried at each turn, store the best scores in a file (Chapter 6), let the player replay without restarting the program, or keep a dictionary of player names and scores (Chapter 5).
