A.5 Aggregate functions: Exercises 1–3

-- Exercise 1: one row per price with the number of apps at that price
SELECT price, COUNT(*)
FROM fake_apps
GROUP BY price;

-- ...only apps downloaded more than 20,000 times
SELECT price, COUNT(*)
FROM fake_apps
WHERE downloads > 20000
GROUP BY price;

-- ...total downloads per category
SELECT category, SUM(downloads)
FROM fake_apps
GROUP BY category;

-- Exercise 2
SELECT category,
   price,
   AVG(downloads)
FROM fake_apps
GROUP BY 1, 2;

-- Exercise 3
SELECT price,
   ROUND(AVG(downloads)),
   COUNT(*)
FROM fake_apps
GROUP BY price
HAVING COUNT(*) > 10;

A.6 Project 5

Note: The exercise text calls the table startups, but the table in the database file is named startup.
-- Warm-up
SELECT COUNT(*) FROM startup;
SELECT SUM(valuation) FROM startup;
SELECT MAX(raised) FROM startup;
SELECT MAX(raised) FROM startup WHERE stage = 'Seed';
SELECT MIN(founded) FROM startup;

-- Valuations among sectors
SELECT AVG(valuation) FROM startup;

SELECT category, AVG(valuation)
FROM startup
GROUP BY category;

SELECT category, ROUND(AVG(valuation), 2)
FROM startup
GROUP BY category;

SELECT category, ROUND(AVG(valuation), 2)
FROM startup
GROUP BY category
ORDER BY 2 DESC;

-- Most competitive markets
SELECT category, COUNT(*)
FROM startup
GROUP BY category;

SELECT category, COUNT(*)
FROM startup
GROUP BY category
HAVING COUNT(*) > 3;

-- Size by location
SELECT location, AVG(employees)
FROM startup
GROUP BY location;

SELECT location, AVG(employees)
FROM startup
GROUP BY location
HAVING AVG(employees) > 500;

A.7 Project 6

-- Top five stories
SELECT title, score
FROM hacker_news
ORDER BY score DESC
LIMIT 5;

-- Total score of all stories
SELECT SUM(score) FROM hacker_news;

-- Users with a combined score above 200
SELECT user, SUM(score)
FROM hacker_news
GROUP BY user
HAVING SUM(score) > 200
ORDER BY 2 DESC;

-- Rickroll offenders
SELECT user, COUNT(*)
FROM hacker_news
WHERE url LIKE '%watch?v=dQw4w9WgXcQ%'
GROUP BY user
ORDER BY 2 DESC;

-- Which sites feed Hacker News?
SELECT CASE
   WHEN url LIKE '%github.com%'  THEN 'GitHub'
   WHEN url LIKE '%medium.com%'  THEN 'Medium'
   WHEN url LIKE '%nytimes.com%' THEN 'New York Times'
   ELSE 'Other'
  END AS 'Source',
  COUNT(*)
FROM hacker_news
GROUP BY 1;

-- Hour of the day, average score, number of stories
SELECT strftime('%H', timestamp),
   AVG(score),
   COUNT(*)
FROM hacker_news
GROUP BY 1;

-- Cleaner version
SELECT strftime('%H', timestamp) AS 'Hour',
   ROUND(AVG(score), 1) AS 'Average Score',
   COUNT(*) AS 'Number of Stories'
FROM hacker_news
WHERE timestamp IS NOT NULL
GROUP BY 1
ORDER BY 2 DESC;

A.8 Multiple tables: Exercises 1–8

-- Exercise 1: look at the tables, then find the answers by reading them
SELECT * FROM orders LIMIT 5;
SELECT * FROM subscriptions LIMIT 5;
SELECT * FROM customers LIMIT 5;

-- Exercise 2
SELECT *
FROM orders
JOIN subscriptions
  ON orders.subscription_id = subscriptions.subscription_id;

SELECT *
FROM orders
JOIN subscriptions
  ON orders.subscription_id = subscriptions.subscription_id
WHERE subscriptions.description = 'Fashion Magazine';

-- Exercise 3
SELECT COUNT(*) FROM online;

SELECT COUNT(*)
FROM newspaper
JOIN online
  ON newspaper.id = online.id;

-- Exercise 4
SELECT *
FROM newspaper
LEFT JOIN online
  ON newspaper.id = online.id;

SELECT *
FROM newspaper
LEFT JOIN online
  ON newspaper.id = online.id
WHERE online.id IS NULL;

-- Exercise 5
SELECT *
FROM classes
JOIN students
  ON classes.id = students.class_id;

-- Exercise 6
SELECT COUNT(*)
FROM newspaper
WHERE start_month <= 3
  AND end_month >= 3;

SELECT *
FROM newspaper
CROSS JOIN months;

SELECT *
FROM newspaper
CROSS JOIN months
WHERE start_month <= month
  AND end_month >= month;

SELECT month,
  COUNT(*)
FROM newspaper
CROSS JOIN months
WHERE start_month <= month AND end_month >= month
GROUP BY month;

-- Exercise 7
SELECT * FROM newspaper
UNION
SELECT * FROM online;

-- Exercise 8
WITH previous_query AS (
   SELECT customer_id,
      COUNT(subscription_id) AS 'subscriptions'
   FROM orders
   GROUP BY customer_id
)
SELECT customers.customer_name,
   previous_query.subscriptions
FROM previous_query
JOIN customers
  ON previous_query.customer_id = customers.customer_id;

A.9 Project 7

Note: Assumption: the foreign keys are named rider_id and car_id in trips, and the other columns are named cost, total_trips, status and trips_completed. Verify the real names with SELECT * FROM table_name; and adapt the queries if needed.
-- 1 and 2: look at the tables (primary key is id in all three)
SELECT * FROM trips;
SELECT * FROM riders;
SELECT * FROM cars;

-- 3: every rider paired with every car (not useful: no real relationship)
SELECT *
FROM riders
CROSS JOIN cars;

-- 4: trip log (trips is the left table)
SELECT *
FROM trips
LEFT JOIN riders
  ON trips.rider_id = riders.id;

-- 5: trips linked to the cars used
SELECT *
FROM trips
JOIN cars
  ON trips.car_id = cars.id;

-- 6: stack riders on top of riders2
SELECT * FROM riders
UNION
SELECT * FROM riders2;

-- 7: average cost of a trip
SELECT ROUND(AVG(cost), 2) FROM trips;

-- 8: irregular users
SELECT * FROM riders
WHERE total_trips < 500;

-- 9: number of active cars
SELECT COUNT(*) FROM cars
WHERE status = 'active';

-- 10: the two cars with the most trips completed
SELECT * FROM cars
ORDER BY trips_completed DESC
LIMIT 2;
