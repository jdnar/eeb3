B.1 Introduction

  • 1.1 The table has 3 columns (id, name, age) and 2 rows. Suitable types: id INTEGER, name TEXT, age INTEGER.
  • 1.2 The clause is CREATE TABLE. The table name is cars. The parameter is the list of columns and types in parentheses: (brand TEXT, year INTEGER). The semicolon marks the end of the statement.
  • 1.3 Price: REAL. Birthday: DATE. Number of children: INTEGER. City: TEXT.

B.2 Manipulation

-- 2.1
CREATE TABLE books (
   id INTEGER,
   title TEXT,
   author TEXT,
   year INTEGER
);

-- 2.2
INSERT INTO books (id, title, author, year) VALUES (1, 'Dune', 'Frank Herbert', 1965);
INSERT INTO books (id, title, author, year) VALUES (2, 'Emma', 'Jane Austen', 1815);
INSERT INTO books (id, title, author, year) VALUES (3, 'Ulysses', 'James Joyce', 1922);

-- 2.3
SELECT title FROM books;

-- and all the columns:
SELECT * FROM books;

-- 2.4
ALTER TABLE books
ADD COLUMN rating REAL;

SELECT * FROM books;

-- 2.5
UPDATE books
SET rating = 4.5
WHERE id = 2;

SELECT * FROM books;

-- 2.6
DELETE FROM books
WHERE rating IS NULL;

SELECT * FROM books;

-- 2.7
CREATE TABLE members (
   id INTEGER PRIMARY KEY,
   email TEXT UNIQUE,
   name TEXT NOT NULL,
   country TEXT DEFAULT 'Belgium'
);

INSERT INTO members (id, email, name)
VALUES (1, 'ana@example.com', 'Ana');

SELECT * FROM members;

-- These two are rejected by the constraints:
-- INSERT INTO members (id, email, name) VALUES (1, 'b@example.com', 'Ben');   (PRIMARY KEY violated)
-- INSERT INTO members (id, email) VALUES (2, 'c@example.com');               (NOT NULL violated)

B.3 Queries

Note: Warm-ups 3, 4 and 5 use the practice database created at the start of Warm-up 3.
-- 3.1
SELECT name, species FROM pets;

-- 3.2
SELECT name AS 'Pet name' FROM pets;

-- 3.3
SELECT DISTINCT species FROM pets;

-- 3.4
SELECT name, age FROM pets WHERE age > 5;

-- 3.5
SELECT name FROM pets WHERE name LIKE 'M%';

-- second part:
SELECT name FROM pets WHERE name LIKE 'R_x';

-- 3.6
SELECT name FROM pets WHERE owner_id IS NULL;

-- 3.7
SELECT name, age FROM pets WHERE age BETWEEN 2 AND 4;

-- 3.8
SELECT name FROM pets WHERE species = 'dog' AND age < 5;

-- 3.9
SELECT name FROM pets WHERE species = 'fish' OR species = 'bird';

-- 3.10
SELECT name, weight FROM pets ORDER BY weight DESC;

-- 3.11
SELECT name, age FROM pets ORDER BY age DESC LIMIT 3;

-- 3.12
SELECT name,
CASE
  WHEN weight > 15 THEN 'Large'
  WHEN weight > 3  THEN 'Medium'
  ELSE 'Small'
END AS 'size'
FROM pets;

B.4 Aggregate Functions

-- 4.1
SELECT COUNT(*) FROM pets;

-- 4.2
SELECT SUM(age) FROM pets;

-- 4.3
SELECT MAX(age), MIN(age) FROM pets;

-- 4.4
SELECT AVG(age) FROM pets;

-- 4.5
SELECT ROUND(AVG(weight), 1) FROM pets;

-- 4.6
SELECT species, COUNT(*)
FROM pets
GROUP BY species;

-- 4.7
SELECT species, ROUND(AVG(age), 1)
FROM pets
GROUP BY 1
ORDER BY 2 DESC;

-- 4.8
SELECT species, COUNT(*)
FROM pets
GROUP BY species
HAVING COUNT(*) > 1;

B.5 Multiple Tables

-- 5.1
SELECT pets.name AS pet, owners.name AS owner
FROM pets
JOIN owners
  ON pets.owner_id = owners.id
ORDER BY pets.id;

-- 5.2
SELECT COUNT(*)
FROM pets
JOIN owners
  ON pets.owner_id = owners.id;

-- 5.3
SELECT pets.name AS pet, owners.name AS owner
FROM pets
LEFT JOIN owners
  ON pets.owner_id = owners.id
ORDER BY pets.id;

-- 5.4
SELECT pets.name
FROM pets
LEFT JOIN owners
  ON pets.owner_id = owners.id
WHERE owners.id IS NULL;

-- 5.5
-- Primary key of owners: id
-- Foreign key in pets: owner_id (it refers to owners.id)

-- 5.6
CREATE TABLE days (day TEXT);
INSERT INTO days (day) VALUES ('Mon');
INSERT INTO days (day) VALUES ('Tue');

SELECT owners.name, days.day
FROM owners
CROSS JOIN days
ORDER BY owners.id, days.day;

-- 5.7
SELECT name FROM owners
UNION
SELECT name FROM pets;

-- 5.8
WITH pet_counts AS (
   SELECT owner_id, COUNT(*) AS pets
   FROM pets
   GROUP BY owner_id
)
SELECT owners.name, pet_counts.pets
FROM pet_counts
JOIN owners
  ON pet_counts.owner_id = owners.id
ORDER BY owners.name;
