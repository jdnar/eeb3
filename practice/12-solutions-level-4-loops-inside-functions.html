Exercise 25 — Print the numbers

def print_numbers(n):
    for i in range(1, n + 1):
        print(i)

# Main program
print_numbers(5)

Exercise 26 — Countdown

def countdown(n):
    while n >= 1:
        print(n)
        n -= 1
    print("Liftoff!")

# Main program
countdown(5)

Exercise 27 — Sum from 1 to n

def sum_to(n):
    total = 0
    for number in range(1, n + 1):
        total += number
    return total

Exercise 28 — Multiplication table

def multiplication_table(n):
    for i in range(1, 11):
        print(n, "x", i, "=", n * i)

# Main program
multiplication_table(7)

Exercise 29 — Factorial

def factorial(n):
    result = 1
    for i in range(1, n + 1):
        result *= i
    return result

Exercise 30 — Draw a rectangle

def draw_rectangle(width, height):
    for row in range(height):
        print("X" * width)

# Main program
draw_rectangle(6, 3)

Exercise 31 — Count the divisors

def count_divisors(n):
    count = 0
    for i in range(1, n + 1):
        if n % i == 0:
            count += 1
    return count

Exercise 32 — Prime numbers

def count_divisors(n):
    count = 0
    for i in range(1, n + 1):
        if n % i == 0:
            count += 1
    return count

def is_prime(n):
    return count_divisors(n) == 2

Exercise 33 — Sum of the even numbers

def sum_of_evens(n):
    total = 0
    for number in range(2, n + 1, 2):
        total += number
    return total

Exercise 34 — Power without **

def power(base, exponent):
    result = 1
    for i in range(exponent):
        result *= base
    return result
