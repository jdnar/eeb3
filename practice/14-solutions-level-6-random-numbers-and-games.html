Exercise 41 — Roll a die

import random

def roll_die():
    return random.randint(1, 6)

# Main program
print("You rolled", roll_die())

Exercise 42 — Flip a coin

import random

def flip_coin():
    return random.choice(["heads", "tails"])

# Main program
print(flip_coin())

Exercise 43 — Roll several dice

import random

def roll_die():
    return random.randint(1, 6)

def roll_dice(n):
    total = 0
    for i in range(n):
        total += roll_die()
    return total

# Main program
print("Two dice:", roll_dice(2))

Exercise 44 — Too low, too high

def check_guess(secret, guess):
    if guess < secret:
        return "too low"
    elif guess > secret:
        return "too high"
    else:
        return "correct"

Exercise 45 — Who wins?

def winner(player, computer):
    if player == computer:
        return "draw"
    elif (player == "rock" and computer == "scissors") or (player == "scissors" and computer == "paper") or (player == "paper" and computer == "rock"):
        return "player"
    else:
        return "computer"

Exercise 46 — The computer’s move

import random

def random_move():
    return random.choice(["rock", "paper", "scissors"])

# Main program
print("The computer chose", random_move())
